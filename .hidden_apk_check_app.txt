package com.example.apkcheck

import android.app.Activity
import android.content.Intent
import android.net.Uri
import android.os.Bundle
import android.provider.OpenableColumns
import android.widget.Toast
import androidx.activity.result.contract.ActivityResultContracts
import androidx.appcompat.app.AppCompatActivity
import java.io.File
import java.io.FileOutputStream

class MainActivity : AppCompatActivity() {
    private val requestFile by lazy {
        registerForActivityResult(ActivityResultContracts.GetContent()) { uri: Uri? ->
            uri?.let { checkApk(it) }
        }
    }

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        requestFile.launch("application/vnd.android.package-archive")
    }

    private fun checkApk(uri: Uri) {
        try {
            val size = contentResolver.openFileDescriptor(uri, "r")?.statSize ?: -1L
            val name = getName(uri)
            val mgr = packageManager
            val info = mgr.getPackageArchiveInfo(uri.toString(), 0)
            val label = info?.applicationInfo?.loadLabel(mgr)?.toString() ?: "?"
            val hashCode = uri.lastPathSegment?.hashCode() ?: 0
            val outPath = getExternalFilesDir(null)?.resolve("apk_check_${hashCode}.apk")
            outPath?.let {
                contentResolver.openInputStream(uri)?.use { input ->
                    FileOutputStream(it).use { output ->
                        input.copyTo(output)
                    }
                }
                Toast.makeText(
                    this,
                    "name=${name}\nsize=$size\nlabel=$label\ncopied=${it.exists()}",
                    Toast.LENGTH_LONG
                ).show()
            } ?: Toast.makeText(this, "no external dir", Toast.LENGTH_SHORT).show()
        } catch (e: Exception) {
            Toast.makeText(this, "error: ${e.message}", Toast.LENGTH_LONG).show()
        }
    }

    private fun getName(uri: Uri): String {
        var result = "unknown"
        contentResolver.query(uri, null, null, null, null)?.use { cursor ->
            val nameIndex = cursor.getColumnIndex(OpenableColumns.DISPLAY_NAME)
            if (cursor.moveToFirst()) {
                result = cursor.getString(nameIndex)
            }
        }
        return result
    }
}
